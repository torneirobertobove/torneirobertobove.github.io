async function caricaProfilo(){ const { data:{ user } } = await client.auth.getUser(); if(!user){ window.location.href="index.html"; return; } userId=user.id; emailUtente=user.email; document.getElementById( "emailBox" ).innerHTML= "Email: " + emailUtente; const { data, error } = await client .from("profili") .select("*") .eq( "user_id", userId ) .single(); if(error && error.code !== "PGRST116"){ console.error( "Errore caricamento profilo:", error ); document.getElementById( "status" ).innerHTML= "Errore caricamento profilo"; return; } if(!data){ console.log( "Profilo non presente, creo record iniziale" ); const { error:erroreCreazione } = await client .from("profili") .insert({ user_id:userId, email:emailUtente, nome:"", cognome:"", telefono:"", livello:"", alias:"", note:"", updated_at: new Date() .toISOString() }); if(erroreCreazione){ console.error( "Errore creazione profilo:", erroreCreazione ); document.getElementById( "status" ).innerHTML= "Errore creazione profilo"; return; } return; } document.getElementById("nome").value= data.nome || ""; document.getElementById("cognome").value= data.cognome || ""; document.getElementById("telefono").value= data.telefono || ""; document.getElementById("livello").value= data.livello || ""; document.getElementById("alias").value= data.alias || ""; document.getElementById("note").value= data.note || ""; }
