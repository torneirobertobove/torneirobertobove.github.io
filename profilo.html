async function caricaProfilo(){ const { data:{ user } } = await sb.auth.getUser(); if(!user){ window.location.href="index.html"; return; } userId=user.id; emailUtente=user.email; const emailBox= document.getElementById( "emailBox" ); if(emailBox){ emailBox.innerHTML= "Email: " + emailUtente; } const { data, error } = await sb .from("profili") .select("*") .eq( "user_id", userId ) .single(); if(error && error.code !== "PGRST116"){ console.error( "Errore caricamento profilo:", error ); const status= document.getElementById( "status" ); if(status){ status.innerHTML= "Errore caricamento profilo"; } return; } if(!data){ console.log( "Profilo non presente, creo record iniziale" ); const { error:erroreCreazione } = await sb .from("profili") .insert({ user_id:userId, email:emailUtente, nome:"", cognome:"", telefono:"", livello:"", alias:"", note:"", ruolo:"user", updated_at: new Date() .toISOString() }); if(erroreCreazione){ console.error( "Errore creazione profilo:", erroreCreazione ); const status= document.getElementById( "status" ); if(status){ status.innerHTML= "Errore creazione profilo"; } return; } return; } const campi=[ "nome", "cognome", "telefono", "livello", "alias", "note" ]; campi.forEach(campo=>{ const elemento= document.getElementById( campo ); if(elemento){ elemento.value= data[campo] || ""; } }); }
