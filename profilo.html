async function caricaProfilo(){ const { data:{ user } } = await sb.auth.getUser(); if(!user){ window.location.href="index.html"; return; } userId=user.id; emailUtente=user.email; const emailBox= document.getElementById( "emailBox" ); if(emailBox){ emailBox.innerHTML= "Email: " + emailUtente; } const { data, error } = await sb .from("profili") .select("*") .eq( "user_id", userId ) .single(); if(error && error.code !== "PGRST116"){ console.error( "Errore caricamento profilo:", error ); const status= document.getElementById( "status" ); if(status){ status.innerHTML= "Errore caricamento profilo"; } return; } if(!data){ console.log( "Profilo non presente, creo record iniziale" ); const { error:erroreCreazione } = await sb .from("profili") .insert({ user_id:userId, email:emailUtente, nome:"", cognome:"", ruolo:"user" }); if(erroreCreazione){ console.error( "Errore creazione profilo:", erroreCreazione ); const status= document.getElementById( "status" ); if(status){ status.innerHTML= "Errore creazione profilo"; } return; } return; } const nome= document.getElementById( "nome" ); const cognome= document.getElementById( "cognome" ); if(nome){ nome.value= data.nome || ""; } if(cognome){ cognome.value= data.cognome || ""; } const telefono= document.getElementById( "telefono" ); if(telefono){ telefono.value= data.telefono || ""; } const livello= document.getElementById( "livello" ); if(livello){ livello.value= data.livello || ""; } const alias= document.getElementById( "alias" ); if(alias){ alias.value= data.alias || ""; } const note= document.getElementById( "note" ); if(note){ note.value= data.note || ""; } }
