async function creaNuovoTorneo(){ const nome= document .getElementById("adminNomeTorneo") .value .trim(); const data= document .getElementById("adminDataTorneo") .value; const posti= Number( document .getElementById("adminPosti") .value ); const formula= document .getElementById("adminTipoTorneo") .value; const descrizione= document .getElementById("adminDescrizione") .value .trim(); if(!nome){ alert( "Inserire nome torneo" ); return; } if(!posti || posti<2){ alert( "Numero posti non valido" ); return; } /* ADMIN CREA SOLO IL CONTENITORE TORNEO La configurazione completa viene gestita da Bove.html */ const configurazione={ origine:"admin", formulaAdmin:formula, boveConfigurato:false, rules:{}, giocatori:[], squadre:[], gironi:{}, calendario:[], classifiche:{}, faseFinale:{} }; const { data:nuovoTorneo, error }=await sb .from("tornei") .insert({ nome:nome, data:data || null, descrizione:descrizione, posti:posti, stato:"bozza", pubblicato:false, iscrizioni_chiuse:false, configurazione:configurazione }) .select("*") .single(); if(error){ console.error( "Errore creazione torneo:", error ); alert( "Errore creazione torneo: " + error.message ); return; } /* RICARICA LISTA TORNEI */ await caricaTorneiSupabase(); adminState.torneoSelezionato= nuovoTorneo.id; salvaAdminState(); renderAdmin(); /* CREA LINK DIRETTO A BOVE */ const link= window.location.origin + window.location.pathname.replace( "admin.html", "" ) + "Bove.html?idTorneo=" + nuovoTorneo.id; const campo= document.getElementById( "linkBoveGenerato" ); if(campo){ campo.value= link; } alert( "✅ Torneo creato correttamente. Ora puoi configurarlo in Bove.html" ); }
